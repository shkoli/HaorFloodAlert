# utils/forecast_utils.py
import requests
from datetime import datetime, timedelta

def get_forecast_precip_next_12h(lat=24.87, lon=91.45):
    now = datetime.utcnow()
    url = (
        f"https://api.open-meteo.com/v1/forecast"
        f"?latitude={lat}&longitude={lon}"
        f"&hourly=precipitation"
        f"&forecast_days=2&timezone=Asia/Dhaka"
    )
    try:
        r = requests.get(url, timeout=15).json()
        if 'hourly' not in r or 'precipitation' not in r['hourly']:
            return 0.0
        
        precip_list = r['hourly']['precipitation']
        time_list = r['hourly']['time']
        
        next_12h_sum = 0.0
        for t, p in zip(time_list, precip_list):
            dt = datetime.fromisoformat(t)
            hours_ahead = (dt - now).total_seconds() / 3600
            if 0 < hours_ahead <= 12:
                next_12h_sum += p
        
        return round(next_12h_sum, 1)
    except Exception as e:
        print(f"Forecast API error: {e}")
        return 0.0
